export const ZapiBadge = ({ value, lang = 'pt' }) => {
  const T = ZAPI_CHANNELS_TEXT[lang] || ZAPI_CHANNELS_TEXT.pt;
  const color = { y: 'green', p: 'yellow', n: 'gray', s: 'gray' }[value] || 'gray';
  return (
    <Badge color={color} size="sm" stroke={value === 's'}>
      {T.values[value]}
    </Badge>
  );
};

export const ZapiChannels = ({ route, lang = 'pt' }) => {
  const T = ZAPI_CHANNELS_TEXT[lang] || ZAPI_CHANNELS_TEXT.pt;
  const r = ZAPI_CHANNELS.routes[route];
  if (!r) {
    return <p><strong>{T.missing}</strong></p>;
  }
  const vals = [r[4], r[5], r[6]];
  return (
    <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: '8px 24px', margin: '0 0 24px' }}>
      {T.channels.map((name, i) => (
        <span key={name} style={{ display: 'inline-flex', alignItems: 'center', gap: '8px' }}>
          <span>{name}</span>
          <ZapiBadge value={vals[i]} lang={lang} />
        </span>
      ))}
    </div>
  );
};

export const ZapiChannelMatrix = ({ family, lang = 'pt' }) => {
  const T = ZAPI_CHANNELS_TEXT[lang] || ZAPI_CHANNELS_TEXT.pt;
  const li = { pt: 0, en: 1, es: 2 }[lang] ?? 0;
  const rows = Object.entries(ZAPI_CHANNELS.routes).filter(([, r]) => r[0] === family);
  const subs = ZAPI_CHANNELS.subs[family];
  const blocks = subs
    ? subs.map((s, k) => ({ title: s[li], rows: rows.filter(([, r]) => r[1] === k) }))
    : [{ title: null, rows }];
  const center = { textAlign: 'center' };
  return (
    <div>
      {blocks.map((b, bi) => {
        const hasNotes = b.rows.some(([slug]) => ZAPI_CHANNELS.notes[slug]);
        return (
        <div key={bi}>
          {b.title && <p><strong>{b.title}</strong></p>}
          <table>
            <thead>
              <tr>
                <th>{T.route}</th>
                {T.channels.map((n) => <th key={n} style={center}>{n}</th>)}
                {hasNotes && <th>{T.note}</th>}
              </tr>
            </thead>
            <tbody>
              {b.rows.map(([slug, r]) => (
                <tr key={slug}>
                  <td><a href={T.base + '/z-api/' + r[0] + '/' + slug}><code>{r[2]} {r[3]}</code></a></td>
                  {[r[4], r[5], r[6]].map((v, i) => (
                    <td key={i} style={center}><ZapiBadge value={v} lang={lang} /></td>
                  ))}
                  {hasNotes && <td>{ZAPI_CHANNELS.notes[slug] ? ZAPI_CHANNELS.notes[slug][li] : ''}</td>}
                </tr>
              ))}
            </tbody>
          </table>
        </div>
        );
      })}
    </div>
  );
};
