export const ChannelBadge = ({ value, lang = 'pt' }) => {
  const T = CHANNEL_TEXT[lang] || CHANNEL_TEXT.pt;
  const color = { yes: 'green', partial: 'yellow', no: 'gray', soon: 'gray', dev: 'gray', unknown: 'gray' }[value] || 'gray';
  const outline = value === 'soon' || value === 'dev' || value === 'unknown';
  return (
    <Badge color={color} size="sm" stroke={outline}>
      {T.values[value]}
    </Badge>
  );
};

export const ChannelBadges = ({ type, scope, lang = 'pt' }) => {
  const T = CHANNEL_TEXT[lang] || CHANNEL_TEXT.pt;
  const vals = scope ? CHANNEL_SCOPES[scope] : channelValues(type);
  if (!vals) {
    return <p><strong>{T.missing}</strong></p>;
  }
  if (vals.every((v) => v === 'dev')) {
    return null;
  }
  return (
    <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: '8px 16px', margin: '0 0 24px' }}>
      {CHANNEL_NAMES.map((name, i) => (
        <span key={name} style={{ display: 'inline-flex', alignItems: 'center', gap: '8px' }}>
          <span>{name}</span>
          <ChannelBadge value={vals[i]} lang={lang} />
        </span>
      ))}
    </div>
  );
};

export const channelRich = (text) =>
  text
    .split(/(\*\*[^*]+\*\*|`[^`]+`)/g)
    .map((part, i) =>
      part.startsWith('**') ? (
        <strong key={i}>{part.slice(2, -2)}</strong>
      ) : part.startsWith('`') ? (
        <code key={i}>{part.slice(1, -1)}</code>
      ) : (
        part
      ),
    );

export const ChannelNotes = ({ scope, lang = 'pt' }) => {
  const T = CHANNEL_TEXT[lang] || CHANNEL_TEXT.pt;
  const notes = CHANNEL_SCOPE_NOTES[scope] || [];
  if (notes.length === 0) {
    return null;
  }
  const kindValue = { partial: 'partial', unknown: 'unknown', no: 'no' };
  return (
    <div>
      <p><strong>{T.notes}</strong></p>
      <ul>
        {notes.map((note, i) => (
          <li key={i}>
            {kindValue[note[1]] && <ChannelBadge value={kindValue[note[1]]} lang={lang} />} <strong>{CHANNEL_NAMES[note[0]]}</strong>: {channelRich(note[2][lang])}
          </li>
        ))}
      </ul>
    </div>
  );
};

export const ChannelMatrix = ({ family, lang = 'pt' }) => {
  const T = CHANNEL_TEXT[lang] || CHANNEL_TEXT.pt;
  const li = { pt: 0, en: 1, es: 2 }[lang] ?? 0;
  const blocks = CHANNEL_FAMILIES[family] || [];
  const center = { textAlign: 'center' };
  return (
    <div>
      {blocks.map((b, bi) => (
        <div key={bi}>
          {b.title && <p><strong>{b.title[li]}</strong></p>}
          <table>
            <thead>
              <tr>
                <th>{T.resource}</th>
                {CHANNEL_NAMES.map((n) => <th key={n} style={center}>{n}</th>)}
              </tr>
            </thead>
            <tbody>
              {b.rows.map((r) => {
                const vals = typeof r[2] === 'string' ? channelValues(r[2]) : r[2];
                const code = r[3] || (typeof r[2] === 'string' ? r[2] : null);
                return (
                  <tr key={r[0]}>
                    <td style={{ minWidth: '230px' }}>
                      <a href={T.base + '/' + r[0]}>{r[1][li]}</a>
                      {code && <span style={{ whiteSpace: 'nowrap', marginLeft: '8px' }}><code>{code}</code></span>}
                    </td>
                    {vals.every((v) => v === 'dev')
                      ? <td colSpan={5} style={center}><ChannelBadge value="dev" lang={lang} /></td>
                      : vals.map((v, i) => (
                        <td key={i} style={center}><ChannelBadge value={v} lang={lang} /></td>
                      ))}
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>
      ))}
    </div>
  );
};

        pt: 'WhatsApp Web, Instagram e Telegram não têm template, e também não precisam: só a Meta Cloud API tem janela de 24 horas para reabrir. Nesses canais, a API recusa o envio com `422` e `subCode` `TEMPLATE_NOT_SUPPORTED`.',
        en: 'WhatsApp Web, Instagram and Telegram have no template, and they do not need one: only Meta Cloud API has a 24-hour window to reopen. On those channels, the API refuses the send with `422` and `subCode` `TEMPLATE_NOT_SUPPORTED`.',
        es: 'WhatsApp Web, Instagram y Telegram no tienen plantilla, y tampoco la necesitan: sola Meta Cloud API tiene ventana de 24 horas para reabrir. En esos canales, la API rechaza el envío con `422` y `subCode` `TEMPLATE_NOT_SUPPORTED`.',
      }],
    ],
  };

  const L = {
    pt: {
      only: 'Só na Meta Cloud API, e exige a janela de 24 horas aberta.',
      win: 'Na Meta Cloud API exige a janela de 24 horas aberta. Nos outros canais que aceitam este tipo não há janela.',
      more: 'O que cada canal aceita, tipo por tipo',
      href: '/channels/message-support',
      missing: 'Tipo sem linha na matriz de canais.',
      dev: 'Ainda não existe na API, em nenhum canal.',
    },
    en: {
      only: 'Meta Cloud API only, and it requires the 24-hour window to be open.',
      win: 'On Meta Cloud API it requires the 24-hour window to be open. The other channels that accept this type have no window.',
      more: 'What each channel accepts, type by type',
      href: '/en/channels/message-support',
      missing: 'Type missing from the channel matrix.',
      dev: 'It does not exist in the API yet, on any channel.',
    },
    es: {
      only: 'Solo en Meta Cloud API, y exige que la ventana de 24 horas esté abierta.',
      win: 'En Meta Cloud API exige que la ventana de 24 horas esté abierta. Los demás canales que aceptan este tipo no tienen ventana.',
      more: 'Qué acepta cada canal, tipo por tipo',
      href: '/es/channels/message-support',
      missing: 'Tipo sin fila en la matriz de canales.',
      dev: 'Todavía no existe en la API, en ningún canal.',
    },
  }[lang];

  const T = CHANNEL_TEXT[lang] || CHANNEL_TEXT.pt;
  const vals = channelValues(type);
  if (!vals) {
    return <p><strong>{L.missing}</strong></p>;
  }

  const rich = channelRich;

  const kindValue = { partial: 'partial', unknown: 'unknown', no: 'no' };
  const isDev = vals.every((v) => v === 'dev');
  const supported = vals.filter((v) => v !== 'no' && v !== 'dev').length;
  const notes = isDev ? [] : NOTES[type] || [];

  return (
    <div>
      {vals[0] === 'yes' && <p>{supported === 1 ? L.only : L.win}</p>}
      {isDev && <p>{L.dev}</p>}

      {notes.length > 0 && (
        <div>
          <p><strong>{T.notes}</strong></p>
          <ul>
            {notes.map((note, i) => (
              <li key={i}>
                {kindValue[note[1]] && <ChannelBadge value={kindValue[note[1]]} lang={lang} />} <strong>{CHANNEL_NAMES[note[0]]}</strong>: {rich(note[2][lang])}
              </li>
            ))}
          </ul>
        </div>
      )}

      <p><a href={L.href}>{L.more}</a></p>
    </div>
  );
};
